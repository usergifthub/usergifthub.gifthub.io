function drawLine(){ if(linePoints.length < 2){ console.log("Потрібно мінімум 2 точки"); return; } if(polyLine != null){ map.removeLayer(polyLine); polyLine = null; } var coordinates = ""; for(var i = 0; i < linePoints.length; i++){ if(i > 0) coordinates += ";"; coordinates += linePoints[i][1] + "," + linePoints[i][0]; } var url = "https://router.project-osrm.org/route/v1/driving/" + coordinates + "?overview=full&geometries=geojson"; console.log("Маршрут:"); console.log(url); var xhr = new XMLHttpRequest(); xhr.open("GET", url, true); xhr.onreadystatechange = function(){ if(xhr.readyState != 4) return; console.log("OSRM HTTP:", xhr.status); console.log("OSRM:", xhr.responseText); if(xhr.status != 200){ console.log( "Помилка отримання маршруту: " + xhr.status ); return; } try{ var data = JSON.parse(xhr.responseText); if(data.code != "Ok"){ console.log( "OSRM не зміг побудувати маршрут: " + data.code ); return; } if(!data.routes || data.routes.length == 0){ console.log( "OSRM не повернув маршрут" ); return; } var route = data.routes[0]; /* * Малюємо отриманий маршрут */ polyLine = L.geoJSON( route.geometry, { style:{ color:"red", weight:5, opacity:0.9 } } ); polyLine.addTo(map); /* * Автоматично показуємо весь маршрут */ map.fitBounds( polyLine.getBounds(), { padding:[30,30] } ); console.log( "Маршрут побудований!" ); console.log( "Відстань: " + route.distance + " м" ); console.log( "Час: " + route.duration + " сек" ); } catch(e){ console.log( "Помилка JSON: " + e.message ); } }; xhr.onerror = function(){ console.log( "Помилка підключення до сервера OSRM" ); }; xhr.send(null); }
